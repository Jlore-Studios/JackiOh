//! `choose_target_where` by name (#1904): the script passes the name of a face function for
//! `where` (`J.choose_target_where({ step = "picked", scope = { ... }, where = "target_where" })`),
//! called as `face[where](ctx, card)` for each card the scope offers (`card` is nil for a hero) and
//! answering whether to offer it. The closure captures only the card, the face, the name and the
//! site — the `draw_while` pattern — so nothing a script sets survives the call (L8), and no registry
//! holds script state (CLAUDE.md rule 4). A misspelled name fails as the verb builds, naming the
//! card, face and function; a call with no `where` offers everything the scope offers.
//!
//! The engine asks `where` with a shared context (`TargetWhere`), and the effect context's readers
//! take theirs mutably (its `rng_*`), so the function reads a `HookArgs` over the same state: the
//! running card as it stands now (`live_self`), the running face and every shared reader. With no
//! running card (a step re-entered after its Spell left the resolving zone) there is nothing to ask
//! with, and nothing is admitted: the prompt fizzles, as one the scope gives no target does.

use std::sync::Arc;

use indexmap::IndexMap;
use mlua::LuaSerdeExt;
use serde::{Deserialize, Serialize};
use serde_json::Value;

use jackioh_engine::CardInstance;
use jackioh_engine::effects::{TargetScope, TargetWhere};
use jackioh_engine::script::{EffectContext, HookArgs};

use crate::api::EffectCall;
use crate::{LuauError, Site};

/// `choose_target_where`'s arguments: the engine's, with `where` naming the face function that
/// admits each card.
#[derive(Serialize, Deserialize, Clone, Debug)]
#[cfg_attr(feature = "ts", derive(ts_rs::TS))]
#[serde(rename_all = "camelCase")]
pub struct ChooseTargetWhereArgs {
    pub step: String,
    #[serde(default, skip_serializing_if = "Option::is_none")]
    pub scope: Option<TargetScope>,
    #[serde(default, rename = "where", skip_serializing_if = "Option::is_none")]
    pub where_: Option<String>,
    #[serde(default, skip_serializing_if = "Option::is_none")]
    pub prompt: Option<String>,
    #[serde(default, skip_serializing_if = "Option::is_none")]
    pub data: Option<IndexMap<String, Value>>,
}

pub fn build_choose_target_where(
    call: &EffectCall,
    site: Site,
    bytecode: &'static [u8],
) -> Result<jackioh_engine::Effect, LuauError> {
    let args: ChooseTargetWhereArgs = crate::api::typed(call, site)?;
    let where_: Option<TargetWhere> = match args.where_ {
        Some(name) => {
            crate::hooks::check_named_hook(site, bytecode, &name)?;
            Some(Arc::new(move |ctx, card| admits(site, bytecode, &name, ctx, card)))
        }
        None => None,
    };
    Ok(jackioh_engine::effects::choose_target_where(
        jackioh_engine::effects::ChooseTargetWhereArgs {
            step: args.step,
            scope: args.scope,
            where_,
            prompt: args.prompt,
            data: args.data,
        },
    ))
}

/// `face[name](ctx, card)` over a `HookArgs` of `ctx`: whether `card` (a hero, `None`) may be
/// offered. A failure panics naming the card, face and function, since an engine hook has no error
/// to return.
fn admits(
    site: Site,
    bytecode: &'static [u8],
    name: &str,
    ctx: &EffectContext<'_>,
    card: Option<&CardInstance>,
) -> bool {
    let Some(self_) = ctx.live_self() else {
        return false;
    };
    let args = HookArgs {
        state: &*ctx.state,
        self_,
        radiant: ctx.radiant,
    };
    match crate::vm::with_vm(site, bytecode, |lua, face| {
        let func = crate::hooks::face_function(&face, name)?;
        let card = match card {
            Some(card) => lua.to_value(card)?,
            None => mlua::Value::Nil,
        };
        crate::reads::scoped(lua, args, |ctx_ud| {
            let answered: mlua::Value = func.call((ctx_ud, card))?;
            Ok(answered.as_boolean().unwrap_or(false))
        })
    }) {
        Ok(admitted) => admitted,
        Err(error) => panic!("{error}"),
    }
}

verbs! {
    verb!(choose_target_where, ChooseTargetWhereArgs);
}

/// #1904: runs `gap_choose_where`'s cry on `face` as a P1 hand card (with none, `with_self` false)
/// against two enemy units, and answers the prompt it opened, or the error its build gave.
fn run_choose_where(
    face: Face,
    with_self: bool,
) -> (
    Result<Option<jackioh_engine::PendingChoice>, jackioh_luau::LuauError>,
    Vec<String>,
) {
    use jackioh_engine::testkit::{EngineSink, PlayerId, Rng};
    common::ensure_installed();
    let code = fixture!("gap_choose_where");
    load_card("lx-gapwhere-load", code).expect("gap_choose_where loads: target_where is a sub-hook");
    let site = Site {
        card: "lx-gapwhere",
        face,
        hook: "cry",
    };
    let mut state = scenario(json!({
        "seed": "lx-gapwhere",
        "p1": { "hand": ["lx-p-for-each"] },
        "p2": { "field": ["lx-fx-001", "lx-fx-002"] },
    }))
    .state()
    .clone();
    let admitted: Vec<String> = jackioh_engine::active_units_of(&state, PlayerId::P2)
        .iter()
        .filter(|unit| unit.def_id == "lx-fx-002")
        .map(|unit| unit.id.clone())
        .collect();
    let probe = jackioh_engine::zone_cards(&state, PlayerId::P1, jackioh_engine::OffFieldZone::Hand)
        .into_iter()
        .next()
        .expect("a hand card to run as")
        .clone();
    let mut events = Vec::new();
    let mut rng = Rng::new(&state.seed, state.rng_cursor);
    let mut ctx = EffectContext::new(EngineSink::new(&mut state, &mut events, &mut rng), PlayerId::P1);
    if with_self {
        ctx.self_ = Some(probe);
    }
    let effects = match run_hook(site, code, &mut ctx) {
        Ok(effects) => effects,
        Err(error) => return (Err(error), admitted),
    };
    for effect in effects {
        (effect.apply)(&mut ctx);
    }
    drop(ctx);
    (Ok(state.pending.clone()), admitted)
}

## Risks

- **Soundness.** Never hand the `&EffectContext` to `reads::scoped` as the effect context. Its readers take it mutably (`rng_*`), and casting `&` to `&mut` is undefined behaviour. The `HookArgs` built from it is the only context the predicate gets.
- **Determinism and purity.** The closure captures only the site, the bytecode and the name, so it holds no state (rule 4, L8). The predicate reads no rng: `HookArgs` has no rng reader.
- **Typed stub and strict analysis.** CI fails if `jackioh.luau` differs from the generator's output, and it runs `luau-analyze --mode=strict` on it, which is not installed locally. `where` is not a Luau reserved word, so `where: string?` parses. Commit exactly what the generator writes.
- **Felinor Feelings and the engine** are untouched, so replays, golden traces and fuzz cannot move.
- **Failure mode.** An erroring `where` panics, naming the card, face and function, as `draw_while`, `with_kill_credit` and the aura predicate do. Custom cards cannot reach this verb: the kit (`kit.luau`) exposes no `choose_target_where`.

## Open questions

None block the build. The plan settles these decisions; a person may revisit them:

1. **Sub-hook name: `target_where`, not the issue's `admits`.** `target_where` pairs with the verb, as `kill_pairs` and `aura_filter` do with theirs.
2. **Predicate context: `HookArgs`.** Rejected: `ConditionContext`, whose `zone` would be false for a Spell, and `TargetCheckArgs`, which needs a `Selection` the engine does not pass for a hero. Also rejected: changing the engine's `TargetWhere` to `&mut EffectContext`, which is engine work and would give a predicate rng access.
3. **No running card: admit nothing, so the prompt fizzles.** Rejected: panicking, which would crash a legal game state.
4. **Honour the name rather than refuse `where` at load.** The issue offers both, and honouring it is the full fix.
