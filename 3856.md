````markdown
4. **`daily/notify.rs`, the task.** Add this beside the queue:

   fn start(app: &Arc<App>) -> mpsc::Sender<String> {
       let (sender, mut receiver) = mpsc::channel(DAILY_NOTIFY_QUEUE);
       let app = Arc::downgrade(app);
       tokio::spawn(async move {
           while let Some(profile_id) = receiver.recv().await {
               let Some(app) = app.upgrade() else { break };
               // Taken off: a move that lands while it is decided for queues it again.
               lock(&app.daily.notifier.queued).shift_remove(profile_id.as_str());
               if let Err(error) = notify_one(&app, &profile_id, app.daily.now()).await {
                   log_warn("daily.notify_failed", json!({ "profileId": profile_id, "message": error.message }));
               }
               app.daily.notifier.busy.send_modify(|busy| *busy = busy.saturating_sub(1));
           }
       });
       sender
   }
   ```
````
