/// The `work` item of an attack hook (its `step` is `"hook"`) carrying `owed` under `key`: what a pause
/// owes, and the plan a hook's own question parks its rest under.
fn hook_resume(work: &str, key: &str, owed: Value) -> Resume {
    engine_resume(work, "hook", IndexMap::from([(key.to_string(), owed)]))
}

/// One attack hook to run (R113): the card it runs for, the player it resolves as, the work item that
/// resumes it (`work`, with the sequence's owed state `owed` under `key`) and the facts in its `ctx.data`.
struct EngineHook<'a> {
    hook: Hook,
    card: &'a CardInstance,
    controller: PlayerId,
    work: &'a str,
    key: &'a str,
    owed: Value,
    facts: IndexMap<String, Value>,
}

/// Runs an attack hook as all five runners do (R113): the context of `run.card` as `run.controller` with
/// the facts, `paused`'s `exits_from` and `summoned` put back when a question split the hook, then its
/// effects through `prompts::run_resumable_list`, a pause there parking the rest under `run.work` with
/// `run.owed`. Only those two are restored: `prompts::run_resume` also restores `chosen_from` and
/// `self_resolving`, and whether that matters here is left as it was.
fn run_engine_hook(
    sink: &mut EngineSink<'_>,
    run: EngineHook<'_>,
    paused: Option<PausedStep>,
) -> crate::prompts::ListStatus {
    let plan = WorkPlan::new(hook_resume(run.work, run.key, run.owed), run.card.controller);
    let mut ctx = crate::resolve::make_context(
        sink,
        Some(run.card),
        crate::resolve::HookOptions {
            controller: Some(run.controller),
            data: Some(run.facts),
            ..crate::resolve::HookOptions::default()
        },
    );
    if let Some(paused) = &paused {
        if let Some(exits_from) = paused.exits_from {
            ctx.exits_from = Some(exits_from);
        }
        if let Some(summoned) = &paused.summoned {
            ctx.summoned = Some(summoned.clone());
        }
    }
    let effects = (run.hook)(&mut ctx);
    crate::prompts::run_resumable_list(&mut ctx, &plan, effects, paused)
}
