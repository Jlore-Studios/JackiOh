/// What `r2214000_a_strength_zero_miss_makes_far_fewer_beginners_mistakes_than_random` counts over one
/// game's turn starts, per picker, random then miss: the picks that ended the turn while a card could
/// be played, of those with a card to play, and the attacks the greedy baseline scores below standing
/// still, of those with such an attack to make.
#[derive(Default)]
struct MissTally {
    idle: [usize; 2],
    idle_of: usize,
    bad_attacks: [usize; 2],
    bad_of: usize,
}

/// The `MissTally` of the random game `r2214000-rates-{game}`, two picks of each picker at each of its
/// turn starts.
fn miss_tally(game: usize) -> MissTally {
    const DRAWS: usize = 2;
    let mut tally = MissTally::default();
    for (at, state) in random_turn_starts(&format!("r2214000-rates-{game}"))
        .iter()
        .enumerate()
    {
        let seat = state.active;
        let position = format!("r2214000-rates-{game}-{at}");
        let has_play = legal_actions(Pinned::default(), state, seat)
            .iter()
            .any(|body| matches!(body, ActionBody::Play { .. }));
        let candidates = candidate_set(state, seat);
        let det = determinize(
            &redact(state, seat),
            seat,
            &mut Rng::new(&format!("{position}:det"), 0),
            DeterminizeOptions {
                match_shown_cost: Some(false),
            },
        );
        let standing = evaluate(&det, seat, NextSwing::Enemy, &GREEDY_EVAL);
        let bad: Vec<&ActionBody> = candidates
            .iter()
            .zip(one_ply_scores(&det, seat, &candidates))
            .filter(|(body, score)| matches!(body, ActionBody::Attack { .. }) && *score < standing)
            .map(|(body, _)| body)
            .collect();
        if has_play {
            tally.idle_of += DRAWS;
        }
        if !bad.is_empty() {
            tally.bad_of += DRAWS;
        }
        for draw in 0..DRAWS {
            let picks = [
                random_action(state, seat, &mut Rng::new(&format!("{position}:{draw}"), 0)),
                skill_miss_action(state, seat, &mut Rng::new(&format!("{position}:{draw}"), 0)),
            ];
            for (picker, pick) in picks.iter().enumerate() {
                if has_play && *pick == Some(ActionBody::EndTurn) {
                    tally.idle[picker] += 1;
                }
                if pick.as_ref().is_some_and(|pick| bad.contains(&pick)) {
                    tally.bad_attacks[picker] += 1;
                }
            }
        }
    }
    tally
}

    // The games are played side by side, each on a thread of its own: one after another they took
    // most of a minute, the last test `rust (test server)` waited on (#2323). Each count is a sum over
    // the games, the same in any order.
    let tallies: Vec<MissTally> = std::thread::scope(|scope| {
        let games: Vec<_> = (0..GAMES)
            .map(|game| scope.spawn(move || miss_tally(game)))
            .collect();
        games
            .into_iter()
            .map(|game| {
                game.join()
                    .unwrap_or_else(|panic| std::panic::resume_unwind(panic))
            })
            .collect()
    });
    let mut total = MissTally::default();
    for tally in tallies {
        for (sum, count) in total.idle.iter_mut().zip(tally.idle) {
            *sum += count;
        }
        for (sum, count) in total.bad_attacks.iter_mut().zip(tally.bad_attacks) {
            *sum += count;
        }
        total.idle_of += tally.idle_of;
        total.bad_of += tally.bad_of;
    }
    let MissTally {
        idle,
        idle_of,
        bad_attacks,
        bad_of,
    } = total;

## Risks

- **Determinism.** Each game reads only its own seeded `Rng`s (`r2214000-rates-{game}…`), and the totals are sums, so thread order changes nothing. Planning checked this: the threaded version printed the same counts as main.
- **Engine state across threads.** The engine's thread-locals (the replay's catalog, the custom-script host, the pool hold) are set per call and are at their defaults on a fresh thread, as on libtest's own thread. `jackioh_cards::register_all()` is a `OnceLock`, safe from many threads; `honest_win` calls the engine from scoped threads the same way.
- **A failing game must still fail the test.** `resume_unwind` re-raises a game thread's panic on the test thread with its message. Do not replace it with `.unwrap()`, which hides the message.
- **Memory.** 20 games' turn starts are held at once (13–41 states each). Planning measured nothing near a limit.
- **Hidden information, replays, the wire:** none are touched (a test-only change). `rust (classify)` should find no card that plays differently, so no pending fragment is needed.
