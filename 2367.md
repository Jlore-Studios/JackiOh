````markdown
/// #2367: what a child is worth that holds a Call to Chaos its line made and can cast at once: the
/// better of its own `open_score` and the best state a cast of such a card reaches, scored as the beam
/// scores a child (`own_prompt_score` where the cast waits on the seat's own prompt, `open_score`
/// otherwise), each cast one `simulate` of `counter`. A card is the line's when the seat did not hold it
/// at the search's root (`held`, the ids of its hand there) and its definition is tagged Call to Chaos.
/// A Call to Chaos is a roll of its edition's effects (three on its Radiant face, R423) that no hand
/// price reads, so a card whose payoff is the Call it makes (Stinkyscape #149 Wildar, R1882000) read as
/// a spent card and was cut before its Call was cast. Only a score: the child stays where it is, so the
/// beam still casts the Call itself. `None` when the game is over, a prompt is open, the child holds no
/// such card or cannot cast one, or the counter refuses a node: the child keeps its plain evaluation.
pub fn own_chaos_score(
    state: &GameState,
    seat: PlayerId,
    held: &IndexSet<String>,
    counter: &dyn NodeCounter,
) -> Option<f64> {
    if state.result.is_some() || state.pending.is_some() {
        return None;
    }
    let made: Vec<&str> = state.players[seat]
        .hand
        .iter()
        .filter(|card| !held.contains(&card.id))
        .filter(|card| {
            find_def(Some(state), &card.def_id).is_some_and(|def| def.tags.contains(&Tag::CallToChaos))
        })
        .map(|card| card.id.as_str())
        .collect();
    if made.is_empty() {
        return None;
    }
    let casts: Vec<ActionBody> = candidate_set(state, seat)
        .into_iter()
        .filter(|action| {
            matches!(action, ActionBody::Play { instance_id, .. } if made.contains(&instance_id.as_str()))
        })
        .collect();
    if casts.is_empty() {
        return None;
    }
    let mut best = open_score(state, seat, NextSwing::Enemy);
    for cast in &casts {
        let Ok(next) = simulate(state, seat, cast, counter)? else {
            continue;
        };
        let score =
            own_prompt_score(&next, seat, counter).unwrap_or_else(|| open_score(&next, seat, NextSwing::Enemy));
        if score > best {
            best = score;
        }
    }
    Some(best)
}
```
````
