````markdown
## Goal

The server serves the player base's per-card **play rate** at a new public, cached route, `GET /api/stats/play-rates`. It reads the card totals it already folds from every game record (R1044000), so the store and the schema do not change. The web gets two functions:

   #[derive(Serialize, Clone, Debug, PartialEq)]
   #[serde(rename_all = "camelCase")]
   pub struct PlayRate {
       pub id: String,
       pub rate: f64,
   }

   /// R1583000: the body: counts and rates only, naming no game and no player.
   #[derive(Serialize, Clone, Debug, PartialEq)]
   #[serde(rename_all = "camelCase")]
   pub struct PlayRatesResponse {
       /// Every human seat of every live game, on every patch and in every mode.
       pub decks: i64,
       /// Each non-token card at least `PLAY_RATES_MIN_GAMES` of those seats held, in catalog-id order.
       pub cards: Vec<PlayRate>,

   /// never count, and a card's rate is R377's `played` over every seat counted.
   fn play_rates_of(totals: &StatsTotals, defs: &CardDefs) -> PlayRatesResponse {
       let report = card_stats_of(
           totals,
           &CardStatsFilter {
               source: SourceFilter::Live,
               mode: None,
               patch: None,
               pilot: PilotFilter::Human,
           },
       );
       let decks = i64::from(report.decks);
       // A listed card's seats are among `decks`, so `decks` is at least `PLAY_RATES_MIN_GAMES` here.
       let cards = report
           .cards
           .iter()
           .filter(|stat| defs.get(&stat.card).is_some_and(|def| !def.token))
           .filter(|stat| tally_of(&stat.in_deck).0 >= PLAY_RATES_MIN_GAMES)
           .map(|stat| PlayRate {
               id: stat.card.clone(),
               rate: tally_of(&stat.played).0 as f64 / decks as f64,
           })
           .collect();
       PlayRatesResponse { decks, cards }
   }

   type KeptPlayRatesCell = Arc<OnceCell<Arc<PlayRatesResponse>>>;

   /// R1583000: the play rates as last computed, with when; one per `App`. The lock is never held
   /// across a computation.
   #[derive(Default)]
   pub struct KeptPlayRates {
       kept: Mutex<Option<(i64, KeptPlayRatesCell)>>,
   }

   impl KeptPlayRates {
       /// The cell made less than `PLAY_RATES_REFRESH_SECONDS` before `now`, or a new empty one.
       fn cell(&self, now: i64) -> KeptPlayRatesCell {
           let mut kept = lock(&self.kept);
           if let Some((made_at, cell)) = kept.as_ref()
               && now - *made_at < PLAY_RATES_REFRESH_SECONDS * MS_PER_SECOND
           {
               return Arc::clone(cell);
           }
           let cell = Arc::new(OnceCell::new());
           *kept = Some((now, Arc::clone(&cell)));
           cell
       }
   }

   /// GET /api/stats/play-rates (R1583000): read off every patch's live totals once per
   /// `PLAY_RATES_REFRESH_SECONDS` however many ask (a failed load is kept for no one), cacheable as long.
   pub async fn get_play_rates(app: &Arc<App>, _req: Req) -> ApiResult {
       let cell = app.play_rates.cell(now_ms());
       let rates = cell
           .get_or_try_init(|| async {
               let mut tx = app.db.begin(None).await?;
               let totals = tx
                   .stats_totals_load(&StatsTotalsQuery {
                       source: SourceFilter::Live,
                       mode: None,
                       patch: None,
                       drills: false,
                   })
                   .await?;
               tx.commit().await?;
               Ok::<_, ApiError>(Arc::new(play_rates_of(&totals, &app.catalog.defs)))
           })
           .await?;
       Ok(cached_ok(to_json(rates.as_ref())?, PLAY_RATES_REFRESH_SECONDS))
   }

   // R1583000: the Pocket Pick, the card the player plays far more often than the player base does,
   // read off the device's tally (R639) against `GET /api/stats/play-rates`. Presentation only; the
   // request sends nothing of the tally (CLAUDE.md rule 7).

   import { getPlayRates, type PlayRatesResponse } from "../net/api.ts";
   import { POCKET_PICK_MIN_GAMES, POCKET_PICK_MIN_PLAYS } from "./config.ts";
   import type { PlayerStats } from "./model.ts";

   /** The card, the share of the player's games it was played in, and the player base's. */
   export type PocketPick = { readonly id: string; readonly playerRate: number; readonly baseRate: number };

   /** A higher player-to-base ratio (cross-multiplied, so a base rate of 0 ranks first), then a higher player rate, then the lower id. */
   function beats(a: PocketPick, b: PocketPick): boolean {
     const ours = a.playerRate * b.baseRate;
     const theirs = b.playerRate * a.baseRate;
     if (ours !== theirs) return ours > theirs;
     if (a.playerRate !== b.playerRate) return a.playerRate > b.playerRate;
     return a.id < b.id;
   }

   /** R1583000: the Pocket Pick, or null below `POCKET_PICK_MIN_GAMES` games or when no listed card qualifies. */
   export function pocketPick(stats: PlayerStats, rates: PlayRatesResponse): PocketPick | null {
     if (stats.games < POCKET_PICK_MIN_GAMES) return null;
     let best: PocketPick | null = null;
     for (const { id, rate } of rates.cards) {
       const played = stats.cards[id]?.played ?? 0;
       if (played < POCKET_PICK_MIN_PLAYS) continue;
       // R639 counts plays; one copy a deck (L3) makes that a game each but for a card played twice.
       const playerRate = Math.min(played, stats.games) / stats.games;
       if (!(playerRate > rate)) continue;
       const candidate: PocketPick = { id, playerRate, baseRate: rate };
       if (best === null || beats(candidate, best)) best = candidate;
     }
     return best;
   }

   /** R1583000: the Pocket Pick against the server's rates; null when the request fails or the tally is too short to ask. */
   export async function loadPocketPick(stats: PlayerStats, signal?: AbortSignal): Promise<PocketPick | null> {
     if (stats.games < POCKET_PICK_MIN_GAMES) return null;
     try {
       return pocketPick(stats, await getPlayRates(signal));
     } catch {
       return null;
     }
   }
   ```
````
