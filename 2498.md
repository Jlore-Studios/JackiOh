```js
import fs from "node:fs";
import path from "node:path";
import { createRequire } from "node:module";
const repo = path.resolve(process.argv[2]);
const JSON5 = createRequire(path.join(repo, "apps/web/package.json"))("json5");
const read = (file) => fs.readFileSync(path.join(repo, file), "utf8");
const catalog = JSON.parse(read("crates/cards/catalog.json"));
const flavour = JSON.parse(read("crates/cards/flavour.json"));
const chinese = JSON.parse(read("crates/cards/chinese.json"));
const audio = JSON5.parse(read("apps/web/src/audio/card-audio.json5"));
const scalar = (s) =>
  /^\p{L}[^\n\r\t]*$/u.test(s) && !/: |:$| #|\s$/.test(s) && !/^(true|false|null|yes|no|on|off|y|n)$/i.test(s)
    ? s
    : JSON.stringify(s);
const out = [];
for (const [id, card] of Object.entries(catalog)) {
  out.push(`${id}: # ${card.name}`);
  const f = flavour[id] ?? {};
  if (f.flavour !== undefined) out.push(`  flavour: ${scalar(f.flavour)}`);
  if (f.artist !== undefined) out.push(`  artist: ${scalar(f.artist)}`);
  const hooks = Object.entries(audio.cards[id] ?? {}).filter(([, a]) => a.voice !== undefined);
  if (hooks.length > 0) out.push("  voice:");
  for (const [hook, a] of hooks) {
    if (hook !== "trigger") out.push(`    ${hook}: ${scalar(a.text)}`);
    else out.push("    trigger:", ...a.lines.map((line) => `      - ${scalar(line)}`));
  }
  const z = chinese[id];
  if (z !== undefined) {
    out.push("  zh:", ...["name", "base", "radiant"].map((k) => `    ${k}: ${scalar(z[k])}`));
    if (z.previews !== undefined) {
      out.push("    previews:", ...Object.entries(z.previews).map(([en, zh]) => `      ${scalar(en)}: ${scalar(zh)}`));
    }
  }
}
out.push("emotes:");
for (const [portrait, e] of Object.entries(audio.emotes)) {
  out.push(`  ${portrait}:`, ...["greetings", "wellPlayed", "oops", "thanks", "threaten"].map((k) => `    ${k}: ${scalar(e[k])}`));
}
fs.writeFileSync(path.join(repo, "crates/cards/card-text.yaml"), `${out.join("\n")}\n`);
```
