````markdown
   //! R4008000: ... (module doc: counts guesses, not requests; distinct wrong codes; malformed counts nothing;
   //! once spent every lookup is 429, a right code's too; the code's own people are not guessing)
   use std::future::Future;
   use crate::api::http::{ApiError, ApiErrorCode, RateLimiter, Req, account_key, address_key, create_rate_limiter, rate_limited};
   use crate::app::now_ms;
   use crate::config::{CODE_MISS_WINDOW_MS, CODE_MISSES_PER_ACCOUNT_PER_HOUR, CODE_MISSES_PER_ADDRESS_PER_HOUR};
   use crate::duos::rooms::code_of;

   const REFUSED_MESSAGE: &str = "too many wrong codes; try again later";

   pub struct CodeGuesses { per_account: RateLimiter, per_address: RateLimiter, counted: RateLimiter }
   // Default: create_rate_limiter(CODE_MISSES_PER_ACCOUNT_PER_HOUR, CODE_MISS_WINDOW_MS),
   //          create_rate_limiter(CODE_MISSES_PER_ADDRESS_PER_HOUR, CODE_MISS_WINDOW_MS),
   //          counted = create_rate_limiter(1, CODE_MISS_WINDOW_MS)

   impl CodeGuesses {
       pub fn gate(&self, req: &Req, account: &str, own: bool) -> Result<(), ApiError> {
           if own { return Ok(()); }
           let now = now_ms();
           let wait = self.per_account.retry_after_ms(&account_key(account), now)
               .max(self.per_address.retry_after_ms(&address_key(&req.address), now));
           if wait > 0 { return Err(rate_limited(REFUSED_MESSAGE, wait)); }
           Ok(())
       }
       pub fn settle<T>(&self, req: &Req, account: &str, own: bool, outcome: Result<T, ApiError>) -> Result<T, ApiError> {
           if !own && matches!(&outcome, Err(error) if error.code == ApiErrorCode::NotFound) { self.miss(req, account); }
           outcome
       }
       pub async fn lookup<T>(&self, req: &Req, account: &str, door: impl Future<Output = Result<T, ApiError>>) -> Result<T, ApiError> {
           self.gate(req, account, false)?;
           let outcome = door.await;
           self.settle(req, account, false, outcome)
       }
       pub fn wrong(&self, req: &Req, account: &str) -> Result<(), ApiError> {   // for the cancel doors
           self.gate(req, account, false)?; self.miss(req, account); Ok(())
       }
       fn miss(&self, req: &Req, account: &str) {
           let Some(code) = code_of(req) else { return };   // malformed: refused unread, counts nothing
           let now = now_ms();
           for (allowance, key) in [(&self.per_account, account_key(account)), (&self.per_address, address_key(&req.address))] {
               if self.counted.allow(&format!("{key}|{code}"), now) { allowance.allow(&key, now); }   // once per (who, code) per window
           }
       }
   }
   ```

   `counted` is a limit-1 limiter, so a repeated code costs one guess. Counting every 404 would lock a user out, because the lobby screen re-reads a closed lobby every `LOBBY_POLL_MS`. The code's own people skip the gate, so another account at the same address cannot block a member's own reads. Use `RateLimiter`'s existing `allow` and `retry_after_ms`; add nothing to `http.rs`.
````
