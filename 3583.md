       (a < c && b > d) || (a > c && b < d)

   /// R3583000: one act's rows before its boss, from `RUN_MAP_PATHS` walks.
   fn act_spots(rng: &mut Rng) -> Vec<Vec<Spot>> {
       let columns: Vec<usize> = (0..RUN_MAP_COLUMNS).collect();
       let mut walks: Vec<Vec<usize>> = Vec::new();
       for start in rng.shuffle(&columns).into_iter().take(RUN_MAP_PATHS) {
           let mut walk = vec![start];
           for row in 1..RUN_ACT_ROWS {
               let from = walk[row - 1];
               let options: Vec<usize> = [from.checked_sub(1), Some(from), Some(from + 1)]
                   .into_iter()
                   .flatten()
                   .filter(|to| *to < RUN_MAP_COLUMNS)
                   .filter(|to| !walks.iter().any(|w| crosses((w[row - 1], w[row]), (from, *to))))
                   .collect();
               walk.push(*rng.pick(&options).expect("going straight up crosses no edge"));
           }
           walks.push(walk);
       }
       let at = |row: usize| -> Vec<usize> {
           let mut cols: Vec<usize> = walks.iter().map(|w| w[row]).collect();
           cols.sort_unstable();
           cols.dedup();
           cols
       };
       (0..RUN_ACT_ROWS)
           .map(|row| {
               let above = if row + 1 < RUN_ACT_ROWS { at(row + 1) } else { Vec::new() };
               at(row).into_iter().map(|column| {
                   let mut next: Vec<usize> = walks.iter().filter(|w| w[row] == column)
                       .filter_map(|w| w.get(row + 1).and_then(|c| above.iter().position(|a| a == c)))
                       .collect();
                   if row + 1 == RUN_ACT_ROWS { next = vec![0]; } // the boss
                   next.sort_unstable();
                   next.dedup();
                   Spot { column, next, fight: false }
               }).collect()
           })
           .collect()
   }
