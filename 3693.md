/// A card in a pile or a field slot (I16): the zone its own record must hold, and what holds it.
struct Placed<'a> {
    card: &'a CardInstance,
    zone: Zone,
    holder: &'static str,
}

impl Placed<'_> {
    /// Where it sits, for a finding: "p1's graveyard", "p2's unit zone 3".
    fn slot(&self) -> String {
        match self.zone {
            Zone::Field { player, lane, .. } => format!("{player}'s {} {lane}", self.holder),
            _ => format!("{}'s {}", self.zone.player(), self.holder),
        }
    }
}

/// Every card in a pile or a field slot, each with the zone it must record. A card `held` beneath another
/// is in no pile (`Zone::Gone`, R1850003) and is not walked.
fn placed_cards(state: &GameState) -> Vec<Placed<'_>> {
    let mut out = Vec::new();
    for player in PLAYER_IDS {
        let side = &state.players[player];
        for (zone, holder, pile) in [
            (Zone::Hand { player }, "hand", &side.hand),
            (Zone::Library { player }, "library", &side.library),
            (Zone::Graveyard { player }, "graveyard", &side.graveyard),
            (Zone::Exile { player }, "exile", &side.exile),
            (Zone::Resolving { player }, "resolving", &side.resolving),
        ] {
            out.extend(pile.iter().map(|card| Placed { card, zone: zone.clone(), holder }));
        }
        // Lane N is index N - 1 in each of the four field slots (§3.2).
        let on_field = |card, row, index: usize, holder| Placed {
            card,
            zone: Zone::Field { player, row, lane: index as i32 + 1 },
            holder,
        };
        for (index, pile) in side.units.iter().enumerate() {
            out.extend(pile.iter().flatten().map(|card| on_field(card, Row::Units, index, "unit zone")));
        }
        for (index, top) in side.backrow.iter().enumerate() {
            out.extend(top.iter().map(|card| on_field(card, Row::Backrow, index, "backrow zone")));
        }
        for (index, beneath) in side.backrow_piles.iter().flatten().enumerate() {
            out.extend(beneath.iter().map(|card| on_field(card, Row::Backrow, index, "backrow pile")));
        }
        for (index, carried) in side.carried.iter().flatten().enumerate() {
            out.extend(carried.iter().map(|card| on_field(card, Row::Backrow, index, "carried unit")));
        }
    }
    out
}

/// A zone as a finding spells it.

mod i16_every_card_s_record_agrees_with_the_pile_that_holds_it {
    use super::*;
    const KPOP: &str = "core-050"; // Cry: Allure an enemy permanent, a steal at its next turn's start
    const TOWER: &str = "classicplus-033"; // Ivory Tower: a carrier
    const TICKETED: &str = "stinkyscape-009"; // holds a permanent beneath it

    fn i16(g: &Scenario) -> Vec<String> {
        create_invariant_monitor(g.state()).after(&[], g.state()).into_iter()
            .filter(|finding| finding.starts_with("I16")).collect()
    }

    fn board() -> (Scenario, CardInstance) {
        let mut g = scenario(json!({
            "p1": {
                "hand": [KPOP, TICKETED, VANILLA],
                "backrow": [
                    { "def": TOWER, "lane": 2 },
                    { "def": SHEEPISH, "lane": 3, "faceUp": false },
                    { "def": SHEEPISH, "lane": 3, "stack": true },
                ],
                "library": [VANILLA, VANILLA, VANILLA],
                "mana": 10,
            },
            "p2": {
                "hand": [VANILLA],
                "field": [{ "def": VANILLA, "lane": 1 }, { "def": VANILLA, "lane": 2 }],
                "library": [VANILLA, VANILLA, VANILLA],
            },
        }));
        let stolen = g.unit_at(PlayerId::P2, 1);
        let held = g.unit_at(PlayerId::P2, 2);
        g.play(TICKETED, json!({ "zone": 1, "targets": pick::instance(&held) }));
        g.play(KPOP, json!({ "targets": pick::instance(&stolen) }));
        g.end_turn();
        g.end_turn();
        assert_eq!(g.state().active, PlayerId::P1);
        assert_eq!((g.card(&stolen).controller, g.card(&stolen).owner), (PlayerId::P1, PlayerId::P2));
        assert_eq!(g.state().held.iter().flatten().count(), 1);
        let pile = ZoneSlot { player: PlayerId::P1, row: Row::Backrow, lane: 3 };
        assert_eq!(beneath_at(g.state(), pile).len(), 1);
        let tower = ZoneSlot { player: PlayerId::P1, row: Row::Backrow, lane: 2 };
        let mut rider = new_instance(g.state_mut(), VANILLA, PlayerId::P1, Zone::Hand { player: PlayerId::P1 });
