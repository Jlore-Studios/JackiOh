````markdown
   - The core, which the planner compiled and tested:
   ```rust
   use jackioh_engine::testkit::{SetName, preview_sets, previewed_sets, set_ships};
   #[derive(clap::Args, Debug, Clone, Default, PartialEq, Eq)]
   pub struct PreviewArgs {
       /// Open a set that has not shipped for this run, as `fuzz` does (R1420, R1391). Repeatable.
       #[arg(long = "preview", value_name = "SET", value_parser = parse_set)]
       pub sets: Vec<SetName>,
   }
   pub(crate) fn parse_set(text: &str) -> Result<SetName, String> { text.parse() }
   pub(crate) fn with_preview<T: Send>(sets: &[SetName], body: impl FnOnce() -> T + Send) -> anyhow::Result<T> {
       if sets.is_empty() { return Ok(body()); }
       let opened = sets.to_vec();
       let pool = rayon::ThreadPoolBuilder::new()
           // A worker's preview lasts as long as the worker, which ends with the pool and the run.
           .start_handler(move |_| std::mem::forget(preview_sets(&opened)))
           .build()?;
       Ok(pool.install(body))
````
