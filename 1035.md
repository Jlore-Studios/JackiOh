````markdown
/// #1035: one thread's registry overrides (`thread_overrides`), for another thread to take
/// (`install`), so work a test hands to a blocking pool reads the test's own cards. Each is the
/// leaked `&'static` the override already holds: nothing is copied.
#[derive(Clone, Copy)]
pub struct ThreadOverrides {
    catalog: Option<&'static CardDefs>,
    catalog_version: Option<&'static str>,
    scripts: Option<&'static IndexMap<String, CardScripts>>,
    win_rates: Option<&'static crate::win_rates::WinRateTable>,
    shipped: Option<&'static [crate::catalog::ShippedCatalog]>,
}

impl ThreadOverrides {
    /// Sets these overrides on the calling thread in place of its own; one absent is cleared, as
    /// `clear_overrides` clears it.
    pub fn install(self) {
        CATALOG_OVERRIDE.with(|cell| cell.set(self.catalog));
        CATALOG_VERSION_OVERRIDE.with(|cell| cell.set(self.catalog_version));

        SHIPPED_OVERRIDE.with(|cell| cell.set(self.shipped));
    }
}

/// called on the async thread that hands the job over, and answers the setup the job runs first, so a

pub type EngineThread = fn() -> Box<dyn FnOnce() + Send>;

/// Production's `EngineThread`: the blocking pool's thread reads the registered catalog, as the

/// #1035: `run` on tokio's blocking pool, as Duos's rebuild and the bot seat run theirs: a whole
/// log's fold on a runtime worker holds every socket, clock and route queued behind it. `thread`'s
/// setup runs first, then `run` inside `catch_unwind`; `run` sets the engine's per-call thread-locals
/// (`with_shipped`, R767000). `Err` is a panic's text or the join error's. Nothing in a job logs: a
/// test records its own thread's log, so the caller logs what comes back.
pub async fn off_runtime<T: Send + 'static>(
    thread: EngineThread,
    run: impl FnOnce() -> T + Send + 'static,
) -> Result<T, String> {
    let setup = thread();
    tokio::task::spawn_blocking(move || {
        setup();
        catch_unwind(AssertUnwindSafe(run)).map_err(panic_text)
    })
    .await
    .map_err(|error| error.to_string())?
}
```
````
