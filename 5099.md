       if json {

       }

       }];
       lines.extend(changed.notes.iter().cloned());
       lines.extend(changed.warning.iter().map(|warning| format!("warning: {warning}")));
       Ok(lines.join("\n"))
   }

       println!("{}", render(&pull_request_cards(&root), args.json)?);
       Ok(())
