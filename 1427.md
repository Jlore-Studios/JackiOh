````markdown
   type HeldRequest = { url: URL; signal: AbortSignal | undefined; answer: (body: unknown, status?: number) => void };

   /**
    * Stubs `fetch`: a request `holds` picks waits until the test answers it and, as `fetch` does, rejects the
    * moment its signal aborts; any other is answered at once with the provisional card statistics.
    */
   function holdFetch(holds: (url: URL) => boolean): HeldRequest[] {
     const held: HeldRequest[] = [];
     vi.stubGlobal(
       "fetch",
       vi.fn((input: RequestInfo | URL, init?: RequestInit) => {
         const url = new URL(String(input), window.location.origin);
         if (!holds(url)) {
           return Promise.resolve(new Response(JSON.stringify(MOCK_CARD_STATS_PROVISIONAL), { status: 200 }));
         }
         return new Promise<Response>((resolve, reject) => {
           const signal = init?.signal ?? undefined;
           signal?.addEventListener(
             "abort",
             () => reject(new DOMException("The operation was aborted.", "AbortError")),
             { once: true },
           );
           held.push({ url, signal, answer: (body, status = 200) => resolve(new Response(JSON.stringify(body), { status })) });
         });
       }),
     );
     return held;
   }

   function requested(pathname: string): URL[] {
     return vi
       .mocked(fetch)
       .mock.calls.map(([input]) => new URL(String(input), window.location.origin))
       .filter((url) => url.pathname === pathname);
   }
   ```
````
