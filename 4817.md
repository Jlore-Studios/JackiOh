````markdown
/// R4817000: a face swapped where the card stands (Make Radiant, De-Radiant, §5.2) re-reads the Brittle
/// its new running face prints. A count its printed Brittle started is the card's text, so it goes
/// when the new face prints none (S #9 Ticketed made Radiant), in any zone; on the field a card with
/// no count starts the one its new face prints, as an arrival does (S #135.1 Sack made Radiant, R687's
/// face-down trap excepted). A count that stays, given or printed on both faces, keeps its number and
/// its cycle (B3.3 rule 5). A Vanilla card prints nothing on either face, so its swap changes nothing.
pub(crate) fn refit_brittle_to_face(state: &mut GameState, id: &str) {
    let Some(mut card) = find_instance(state, id).cloned() else {
        return;
    };
    if card.vanilla {
        return;
    }
    let prints = printed_brittle_of(state, &card).is_some_and(|printed| printed > 0);
    if card.brittle.is_some_and(|brittle| brittle.printed == Some(true)) && !prints {
        card.brittle = None;
    } else if card.zone.z() == ZoneName::Field {
        start_brittle_on_field(state, &mut card, false);
    }
    if let Some(live) = find_instance_mut(state, id) {
        live.brittle = card.brittle;
    }
}
```
````
