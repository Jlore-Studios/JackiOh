       pub fn failed(&mut self, message: &str) {
           let now = now_ms();
           if self.passes == 0 { self.since = now; }

           if self.alerted || self.passes < LOOP_FAILING_ALERT_PASSES || now - self.since < LOOP_FAILING_ALERT_MS { return; }
           self.alerted = true;
           tracing::error!(event = self.failing_event.as_str(), alert = true, step = self.step,
               passes = self.passes, failingForMs = now - self.since, message = %message);
       }
       pub fn ok(&mut self) {
           if self.alerted {
               tracing::info!(event = self.recovered_event.as_str(), step = self.step,

           }
           self.passes = 0;
           self.alerted = false;
       }
   }
