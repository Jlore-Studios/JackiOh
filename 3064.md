   /// R3064000: what `unit`'s awakening gives its controller, by `prices`, while its face has a wake
   /// trigger. 0 with `Prices::NONE`, for an unpriced card and at a zero `awaken` weight.
   fn awaken_price(state: &GameState, unit: &CardInstance, w: &EvalWeights, prices: &Prices) -> f64 {

   fn settled_after(before: &GameState, step: impl FnOnce(&mut EngineSink<'_>)) -> GameState {

           let mut next = before.clone();
           sync_fused_scripts(&mut next);
           let mut events: Vec<GameEvent> = Vec::new();

   /// One node; `None` when refused, `Some(None)` on a panic (`note_sim_panic`), an open prompt or a finished game.
   fn wake_once(before: &GameState, id: &str, counter: &dyn NodeCounter) -> Option<Option<GameState>> {
       if !counter.take() { return None; }
       match catch_unwind(AssertUnwindSafe(|| woken(before, id))) {
           Ok(after) => Some((after.pending.is_none() && after.result.is_none()).then_some(after)),
           Err(payload) => { note_sim_panic(counter, payload.as_ref()); Some(None) }

   /// before scoring), plus `AI_EVAL.next_turn_mana` per crystal gained over the twin. Signed. Two nodes.
   fn awaken_probe(before: &GameState, unit: &CardInstance, counter: &dyn NodeCounter) -> Option<Option<f64>> {
       let owner = unit.controller;
       let Some(hooked) = wake_once(before, &unit.id, counter)? else { return Some(None); };
       let (twin, printed) = hookless_twin(before, unit);
       let Some(plain) = wake_once(&twin, &unit.id, counter)? else { return Some(None); };
       let plain = with_text_back(plain, &unit.id, &printed);
       let gained = hooked.players[owner].mana.current - plain.players[owner].mana.current;
