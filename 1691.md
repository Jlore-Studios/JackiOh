   function RetryButton({ wait, onRetry, testId, className }: { wait: number; onRetry: () => void; testId: string; className?: string }): ReactElement {
     return (
       <button type="button" className={className} data-testid={testId} disabled={wait > 0} onClick={onRetry}>
         {wait > 0 ? t("Try again in {seconds} s", { seconds: String(wait) }) : t("Try again")}
       </button>
     );
   }

   /** Try again beside a failure said elsewhere. Render it only while that failure shows, so each failure counts its own wait. */
   export function TryAgainButton({ retryAfterMs, onRetry, testId, className }: { retryAfterMs?: number; onRetry: () => void; testId: string; className?: string }): ReactElement {
     const wait = useCountdown(retryAfterMs);
     return <RetryButton wait={wait} onRetry={onRetry} testId={testId} className={className} />;
   }

   /** A failed read: its sentence and the wait left, then Try again (off through the wait) and `children` in one row. */
   export function AccountError({ message, retryAfterMs, onRetry, testId, retryTestId, children }: {
