       inflight: Mutex<IndexMap<K, Arc<tokio::sync::OnceCell<T>>>>,
   }
   impl<K: Hash + Eq + Clone, T: Clone> Flights<K, T> {
       fn new() -> Flights<K, T> { Flights { inflight: Mutex::new(IndexMap::new()) } }
       async fn run<F, Fut>(&self, key: K, call: F) -> T
       where F: FnOnce() -> Fut, Fut: Future<Output = T>,
       {
           let cell = {
               let mut inflight = self.inflight.lock().unwrap_or_else(|poisoned| poisoned.into_inner());
               Arc::clone(inflight.entry(key.clone()).or_insert_with(|| Arc::new(tokio::sync::OnceCell::new())))
           };
           let _leave = LeaveFlight { flights: self, key, cell: &cell };
           cell.get_or_init(call).await.clone()
       }
   }
   struct LeaveFlight<'a, K: Hash + Eq, T> { flights: &'a Flights<K, T>, key: K, cell: &'a Arc<tokio::sync::OnceCell<T>> }
   impl<K: Hash + Eq, T> Drop for LeaveFlight<'_, K, T> {
       fn drop(&mut self) {
           let mut inflight = self.flights.inflight.lock().unwrap_or_else(|poisoned| poisoned.into_inner());
           let ours = inflight.get(&self.key).is_some_and(|held| Arc::ptr_eq(held, self.cell));
           if ours && (self.cell.initialized() || Arc::strong_count(self.cell) == 2) {
               inflight.shift_remove(&self.key);
           }
       }
   }
