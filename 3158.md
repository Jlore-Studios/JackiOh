````markdown
use std::sync::atomic::{AtomicU32, Ordering};
use std::sync::{Mutex, MutexGuard, PoisonError, mpsc};
use std::time::{Duration, Instant};
use indexmap::IndexMap;
use rayon::prelude::*;

#[derive(Default)]
struct Watch {
    running: Mutex<IndexMap<u32, Instant>>,
    done: AtomicU32,
    slowest: Mutex<Option<(u32, Duration)>>,
}

fn lock<T>(mutex: &Mutex<T>) -> MutexGuard<'_, T> {
    mutex.lock().unwrap_or_else(PoisonError::into_inner)
}

/// `run` over seeds `from..from + ran` under rayon, results in seed order, watched as the header says.
pub(crate) fn watched<T: Send>(
    label: &str,
    from: u32,
    ran: u32,
    reproduce: impl Fn(u32) -> String + Sync,
    run: impl Fn(u32) -> T + Sync + Send,
) -> Vec<T> {
    let watch = Watch::default();
    let shared = &watch;
    let reproduce = &reproduce;
    let started = Instant::now();
    let (stop, stopped) = mpsc::channel::<()>();
    let results = std::thread::scope(|scope| {
        scope.spawn(move || {
            while let Err(mpsc::RecvTimeoutError::Timeout) =
                stopped.recv_timeout(Duration::from_secs(HEARTBEAT_SECS))
            {
                let now = Instant::now();
                let ages: Vec<(u32, Duration)> = lock(&shared.running)
                    .iter()
                    .map(|(seed, began)| (*seed, now.duration_since(*began)))
                    .collect();
                let bound = Duration::from_secs(SEED_STALL_BOUND_SECS);
                if let Some(line) = stall_line(label, &ages, bound, reproduce) {
                    eprintln!("{line}");
                    std::process::exit(1);
                }
                let done = shared.done.load(Ordering::Relaxed);
                eprintln!("{}", heartbeat_line(label, done, ran, started.elapsed(), &ages));
            }
        });
        let results: Vec<T> = (from..from + ran)
            .into_par_iter()
            .map(|seed| {
                let began = Instant::now();
                lock(&shared.running).insert(seed, began);
                let result = run(seed);
                lock(&shared.running).swap_remove(&seed);
                shared.done.fetch_add(1, Ordering::Relaxed);
                keep_slowest(&mut lock(&shared.slowest), seed, began.elapsed());
                result
            })
            .collect();
        drop(stop);
        results
    });
    eprintln!("{}", finished_line(label, ran, started.elapsed(), *lock(&watch.slowest)));
    results
}
```

   - Helper functions, each with a doc comment:
     - `fn longest_first(ages: &[(u32, Duration)]) -> Vec<(u32, Duration)>`: a copy sorted by age, longest first; on a tie the lower seed comes first.
     - `fn stall_line(label: &str, ages: &[(u32, Duration)], bound: Duration, reproduce: &dyn Fn(u32) -> String) -> Option<String>`: looks only at the first entry of `longest_first`. It returns `None` unless that age is strictly greater than `bound`. Otherwise it returns:
       `format!("[{label}] seed {seed} has run {}s, past SEED_STALL_BOUND_SECS ({}s): it stalled, so the wave stops here, failing\n  reproduce: {}", age.as_secs(), bound.as_secs(), reproduce(seed))`
     - `fn heartbeat_line(label: &str, done: u32, ran: u32, elapsed: Duration, ages: &[(u32, Duration)]) -> String` returns:
       `"[{label}] {done} of {ran} seeds done in {elapsed_secs}s; running longest: {list}"`
       `{list}` is the first `IN_FLIGHT_SHOWN` entries of `longest_first`, each written `seed {seed} ({secs}s)` and joined by `", "`. It is `none` when no seed is running.
     - `fn keep_slowest(slowest: &mut Option<(u32, Duration)>, seed: u32, took: Duration)`: replaces the kept value when there is none, when `took` is longer, or when `took` is equal and `seed` is lower.
     - `fn finished_line(label: &str, ran: u32, elapsed: Duration, slowest: Option<(u32, Duration)>) -> String` returns:
       `"[{label}] {ran} seeds in {secs}s; slowest: seed {seed}, {ms}ms"`, or `slowest: none` when there is no seed.
````
