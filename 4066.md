    fn r1026_r113_a_deaths_question_in_the_closing_check_still_runs_the_defenders_hook() {

            &mut state,

            &mut state,
            &defender.id,
            slot(P2, Row::Units, 2),
            Default::default(),

            &death_asker.id,
            slot(P2, Row::Units, 1),
            Default::default(),

        let owed: Vec<&str> = paused

        assert!(replays_to(&game.start, &game.log, game.state()));
