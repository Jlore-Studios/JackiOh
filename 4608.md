   /// R4608000: each finalist's mean score. A row holds one finalist's scores, world 0's first. With
   /// `confirm` on, a row whose world 0 won (above `AI_EVAL.win * AI_WON_SCORE_SHARE`), that has other
   /// worlds and that won on none of them is ranked by the mean of those others; every other row
   /// by the mean of all its worlds.
   pub fn finalist_means(rows: &[Vec<f64>], confirm: bool) -> Vec<f64> {
       let won = AI_EVAL.win * AI_WON_SCORE_SHARE;
       rows.iter()
           .map(|row| match row.split_first() {
               Some((first, others))
                   if confirm && *first > won && !others.is_empty() && others.iter().all(|score| *score <= won) =>
               {
                   mean_of(others)
               }
               _ => mean_of(row),
           })
           .collect()
   }
