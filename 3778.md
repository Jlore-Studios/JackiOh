   }

   pub fn top_up_combat_keywords(args: DrawCombatKeywordsArgs) -> Effect {
       Effect::new("topUpCombatKeywords", move |ctx| {
           let key = args.key.clone().unwrap_or_else(|| COMBAT_KEYWORDS_KEY.to_string());
           let Some(me) = ctx.live_self().cloned() else { return; };
           if me.zone.z() != ZoneName::Field { return; }
           let mut kept = drawn_combat_keywords(&me, Some(&key));
           let want = usize::try_from(args.count).unwrap_or(0);
           if kept.len() >= want && me.memory.contains_key(&key) { return; }
           let mut held = held_beside(ctx.sink.state, &me, &kept);
           while kept.len() < want {
               let candidates = candidates_beside(&held);
               let Some(keyword) = ctx.sink.rng.pick(&candidates).cloned() else { break; };
               held.push(keyword.clone());
               kept.push(keyword);
           }
           if let Some(live) = find_instance_mut(ctx.sink.state, &me.id) {
               live.memory.insert(key, serde_json::to_value(&kept).unwrap_or(serde_json::Value::Null));
           }
       })
   }

   pub fn aura_combat_keywords(args: &HookArgs<'_>, number: &str, key: Option<&str>) -> Vec<Keyword> {

       drawn_combat_keywords(args.self_, key).into_iter().take(shown).collect()
