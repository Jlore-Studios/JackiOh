   useLayoutEffect(() => {

   const dispatch = useCallback(
     (body: ActionBody) => {
       if (ONCE_PER_VIEW.has(body.type)) {

         const sent = lastSent.current;
         if (sent !== null && sent.view === latest.current && sent.key === key) return;

     },
