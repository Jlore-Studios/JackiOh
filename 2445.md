STEMS = [
    ("centred", "centered"), ("centring", "centering"), ("centre", "center"),

    r"^(\.github/|\.harness/|\.squishy/|bot/|\.claude/|reviews/|crates/cards/patches/"

    r"|crates/cards/luau/jackioh\.luau$|apps/web/src/cards/art/fixtures/)"

# Wire, persisted or external names.

    "labelledby",  # aria-labelledby

    "StarscapeTravelled", "starscapeTravelled", "starscape_travelled",  # event

# Bare `cancelled` is a wire name (JSON key or value, DB status, serialized field, GitHub

# that is not a comment, `let` or `const`. Never in WIRE_FREE.
WIRE_BEFORE = "`\"'."
WIRE_AFTER = re.compile(r"\s*:(?!:)")
LOCAL = re.compile(r"\b(let|let mut|const|var)\s+$")
WIRE_FREE = ("apps/web/src/audio/", "apps/web/src/cards/fit.ts", "apps/web/src/replay/")
COMMENT = re.compile(r"\s*(//|/\*|\*|--|#(?!\[))")

def american(match):
    i = next(n for n, g in enumerate(match.groups()) if g is not None)
    text, out = match.group(0), STEMS[i][1]
    if text.isupper():
        return out.upper()
    return out[0].upper() + out[1:] if text[0].isupper() else out


def fix_line(path, line):
    if not STEM.search(line):
        return line
    kept = [text for text in KEEP_TEXT if text in line]
    for i, text in enumerate(kept):
        line = line.replace(text, f"\0{i}\0")
    code = path.endswith(CODE) and not COMMENT.match(line)

    def fix_token(match):

            return token
        if token == "cancelled" and not path.startswith(WIRE_FREE):
            before = line[match.start() - 1] if match.start() else " "

                return token
            if code and WIRE_AFTER.match(line, match.end()) and not LOCAL.search(line, 0, match.start()):
                return token
        return STEM.sub(american, token)

    line = TOKEN.sub(fix_token, line)
    for i, text in enumerate(kept):
        line = line.replace(f"\0{i}\0", text)
    return line
