   pub fn settle_cosmetic_bar(start: i64, gained: i64, position: i64) -> (Vec<CosmeticTokenStep>, i64) {

       let mut paid = Vec::new();
       loop {
           let step = cycle_at(position + paid.len() as i64);
           if points < step.bar_size() {
               return (paid, points);
           }
           points -= step.bar_size();
           paid.push(step);
       }
   }
