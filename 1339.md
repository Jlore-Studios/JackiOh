   pub fn roll_tier<T: Copy + Ord>(set: &[(T, u32)], fallback: T, floor: Option<T>, draw: u32) -> Option<T> {
       fallback_weight(set)?;
       if draw >= TIER_WEIGHT_TOTAL {
           return None;
       }
       let mut edge = 0;
       let mut landed = fallback;
       for (tier, weight) in set {
           edge += weight;
           if draw < edge {
               landed = *tier;
               break;
           }
       }
       Some(match floor {
           Some(floor) if landed < floor => floor,
           _ => landed,
       })
   }
