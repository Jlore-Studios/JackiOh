/// R4515000: the best `unit_worth` to the seat among the seat's Units `action` targets;

fn aimed_worth(state: &GameState, seat: PlayerId, action: &ActionBody, prices: &Prices) -> f64 {

```rust
fn ordered_tries(state: &GameState, seat: PlayerId, mut tries: Vec<ActionBody>, spread: bool, prices: &Prices) -> Vec<ActionBody> {
    if !spread {
        // Stable, so each aim keeps legal_actions order.
        tries.sort_by_key(|action| aim_of(state, seat, action));
        return tries;
    }
    let mut ordered: Vec<ActionBody> = Vec::new();          // the hero's tries, then the dealt picks
    let mut groups: Vec<PickGroup> = Vec::new();
    let mut seen: IndexSet<ActionBody> = IndexSet::new();
    let mut waiting: Vec<ActionBody> = Vec::new();          // a pick's later Tribute sets
    let mut rest: Vec<ActionBody> = Vec::new();
    for action in tries {
        match aim_of(state, seat, &action) {
            Aim::SeatHero => ordered.push(action),
            Aim::Other => rest.push(action),
            Aim::SeatCard => {
                if !seen.insert(pick_of(&action)) { waiting.push(action); continue; }
                let source = source_of(&action);
                let worth = aimed_worth(state, seat, &action, prices);
                match groups.iter_mut().find(|group| group.source == source) {
                    Some(group) => group.picks.push((worth, action)),
                    None => groups.push(PickGroup { source, picks: vec![(worth, action)] }),
                }
            }
        }
    }
    for group in &mut groups {
        // Highest worth first; `sort_by` is stable, so equals keep legal order.
        group.picks.sort_by(|a, b| b.0.partial_cmp(&a.0).unwrap_or(Ordering::Equal));
    }
    let deepest = groups.iter().map(|group| group.picks.len()).max().unwrap_or(0);
    for round in 0..deepest {
        ordered.extend(groups.iter().filter_map(|group| group.picks.get(round)).map(|(_, pick)| pick.clone()));
    }
    ordered.extend(waiting);
    ordered.extend(rest);
    ordered
}
```
