pub const FLOOD_WINDOW_MS: i64 = 1000;
pub const CLOSE_POLICY_VIOLATED: u16 = 1008;

fn prune(window: &mut VecDeque<i64>, now: i64, span: i64) {
    while window.front().is_some_and(|at| *at <= now - span) { window.pop_front(); }
}

#[derive(Default)]
pub struct SeatGuard { actions: VecDeque<i64>, rejections: VecDeque<i64>, alerted: bool, malformed: VecDeque<i64> }

impl SeatGuard {
    pub fn action_allowed(&mut self, now: i64) -> bool {
        prune(&mut self.actions, now, FLOOD_WINDOW_MS);
        if self.actions.len() >= MATCH_ACTIONS_PER_SECOND { return false; }
        self.actions.push_back(now);
        true
    }
    pub fn note_rejection(&mut self, now: i64) -> Option<usize> {
        prune(&mut self.rejections, now, REJECTION_ALERT_WINDOW_MS);
        self.rejections.push_back(now);
        if self.rejections.len() > REJECTION_ALERT_THRESHOLD && !self.alerted {
            self.alerted = true;
            Some(self.rejections.len())
        } else { None }
    }
    pub fn note_malformed(&mut self, now: i64) -> bool {
        prune(&mut self.malformed, now, REJECTION_ALERT_WINDOW_MS);
        self.malformed.push_back(now);
        self.malformed.len() > MALFORMED_FRAMES_PER_WINDOW
    }
}

pub fn alert_unusual_rejections(match_id: &str, profile_id: &str, rejected: usize) {
    log_alert("match.rejections.unusual", json!({ "profileId": profile_id, "matchId": match_id, "rejected": rejected, "windowMs": REJECTION_ALERT_WINDOW_MS }));
}

struct GateState<H> { hello: Option<H>, closed: bool, guard: SeatGuard }
struct Gate<H> { state: Mutex<GateState<H>>, heard: Notify, socket: Socket, parse: fn(&str) -> Result<H, String> }

impl<H> Gate<H> {
    fn receive(&self, text: &str) {
        let mut state = lock(&self.state);
        if state.hello.is_some() || state.closed { return; } // B50: nothing is read after the hello
        match (self.parse)(text) {
            Ok(hello) => { state.hello = Some(hello); drop(state); self.heard.notify_one(); }
            Err(reason) => {
                let spent = state.guard.note_malformed(now_ms());
                drop(state); // the close calls `closed` back
                self.socket.send(encode(&error_message(SocketErrorCode::Malformed, &reason, None)));
                if spent { self.socket.close(Some(CLOSE_POLICY_VIOLATED), Some("too many malformed frames")); }
            }
        }
    }
    fn closed(&self) { lock(&self.state).closed = true; self.heard.notify_one(); }
}
