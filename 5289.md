const INDENT: &str = "  ";
fn is_space(c: char) -> bool { matches!(c, ' ' | '\n' | '\r' | '\t') }
fn is_mark(c: char) -> bool { matches!(c, '{' | '}' | '[' | ']' | ',' | ':') }
fn opens(token: &str) -> bool { matches!(token, "{" | "[") }
fn closes(token: &str) -> bool { matches!(token, "}" | "]") }
/// Each mark, each string whole, each other literal; white space dropped.
fn tokens(json: &str) -> Vec<&str> {
    let mut tokens = Vec::new();
    let mut chars = json.char_indices().peekable();
    while let Some((start, c)) = chars.next() {
        if is_space(c) { continue; }
        let mut end = start + c.len_utf8();
        if c == '"' {
            let mut escaped = false;
            for (at, d) in chars.by_ref() {
                end = at + d.len_utf8();
                if escaped { escaped = false } else if d == '\\' { escaped = true } else if d == '"' { break }
            }
        } else if !is_mark(c) {
            while let Some(&(at, d)) = chars.peek() {
                if is_space(d) || is_mark(d) || d == '"' { break; }
                end = at + d.len_utf8();
                chars.next();
            }
        }
        tokens.push(&json[start..end]);
    }
    tokens
}
fn new_line(out: &mut String, level: usize) { out.push('\n'); out.push_str(&INDENT.repeat(level)); }
pub fn layout(json: &str, depth: usize) -> String {
    let tokens = tokens(json);
    let mut out = String::with_capacity(json.len());
    let mut open: Vec<bool> = Vec::new();
    for (at, token) in tokens.iter().enumerate() {
        let rest = &tokens[at + 1..];
        if opens(token) {
            let wraps = open.len() < depth
                && if open.is_empty() {
                    rest.first().is_some_and(|next| !closes(next))
                } else {
                    rest.iter().find(|t| opens(t) || closes(t)).is_some_and(|t| opens(t))
                };
            out.push_str(token);
            open.push(wraps);
            if wraps { new_line(&mut out, open.len()); }
        } else if closes(token) {
            if open.pop() == Some(true) { new_line(&mut out, open.len()); }
            out.push_str(token);
        } else {
            out.push_str(token);
            if open.last() == Some(&true) {
                match *token { "," => new_line(&mut out, open.len()), ":" => out.push(' '), _ => {} }
            }
        }
    }
    out.push('\n');
    out
}
