   pub fn played_type_of(state: &GameState, event: &GameEvent) -> Option<CardType> {
       let (instance_id, def_id, radiant) = match event {
           GameEvent::CardPlayed { instance_id, def_id, .. } => (instance_id, def_id, false),
           GameEvent::CardResolved { instance_id, def_id, radiant, .. } => (instance_id, def_id, *radiant == Some(true)),
           _ => return None,
       };
       Some(match find_instance(state, instance_id) {
           Some(card) => crate::faces::card_type_of(state, card),
           None => crate::faces::card_type_of_face(state, def_id, radiant),
       })
   }
