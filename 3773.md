````markdown
/// I17: what §4.5's state check and §4.4's pipeline leave behind (§2.3, §4.4, §4.5, R63, R69, R2201000).
/// Every state keeps each seat's current and max mana and each card's `damage` at 0 or more (§2.3). A live game
/// with no prompt open and no mulligan (I7's settled condition) has no field unit at layered health 0 or
/// less, an Indestructible one excepted while its max health is above 0 (R69), and no 1v1 hero that `can_lose`
/// at 0 or less. Every `damage` event of an action's events deals more than 0 (R63; an absorbed hit is
/// `damageAbsorbed`, R240, R1362). A Duos seat (`duos` set) is left to the team's pool bound (§13.4): B15's
/// `sync_pools` writes a spent hero's health into the other duel with no state check there.
fn state_check_findings(events: &[GameEvent], state: &GameState) -> Vec<String> {
    let mut found = Vec::new();
    let turn = state.turn;
    for event in events {
        if let GameEvent::Damage { target_id, amount, .. } = event
            && *amount <= 0
        {
            found.push(format!(
                "I17 empty damage: a damage event of {amount} on {target_id} on turn {turn}, though a hit of 0 \
                 emits nothing (§4.4, R63)"
            ));
        }
    }
    for player in PLAYER_IDS {
        let mana = state.players[player].mana;
        if mana.current < 0 || mana.max < 0 {
            found.push(format!(
                "I17 negative mana: {player} holds {} of {} mana on turn {turn} (§2.3)",
                mana.current, mana.max
            ));
        }
    }
    for placed in placed_cards(state) {
        if placed.card.damage < 0 {
            found.push(format!(
                "I17 negative damage: {} ({}) in {} holds damage {} on turn {turn} (§4.4)",
                placed.card.id, placed.card.def_id, placed.slot(), placed.card.damage
            ));
        }
    }
    if state.result.is_some() || state.pending.is_some() || state.mulligan.is_some() {
        return found;
    }
    let auras = AuraTable::new(state);
    for player in PLAYER_IDS {
        for unit in crate::zones::units_on_field_of(state, player) {
            let view = auras.unit_view(unit);
            let stands = has_keyword(&view.keywords, KeywordKind::Indestructible) && view.max_health > 0;
            if view.health <= 0 && !stands {
                found.push(format!(
                    "I17 dead unit left standing: {} is at {} health of {} max on turn {turn} after the state \
                     check (§4.5, R69)",
                    name_of(unit), view.health, view.max_health
                ));
            }
        }
        let side = &state.players[player];
        if side.duos.is_none() && crate::state::can_lose(state, player) && side.hero.health <= 0 {
            found.push(format!(
                "I17 live hero at 0: {player}'s hero has {} health on turn {turn} and the game is not over \
                 (§4.5, R2201000)",
                side.hero.health
            ));
        }
    }
    found
}
```
````
