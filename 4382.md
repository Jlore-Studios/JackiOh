   pub fn card_eras(id: &str) -> Vec<&'static str> {
       if !CATALOG.contains_key(id) { return Vec::new(); }
       let mut eras = Vec::new();
       let mut before: Option<(Option<usize>, Option<usize>)> = None;
       for delta in HISTORY.versions.iter().filter(|delta| delta.version != crate::CATALOG_VERSION) {
           if delta.absent.contains(id) { before = None; continue; }
