import os, re

D = ("crates/cards/src/scripts", "crates/cards/tests", "crates/engine/tests/rules")
squash = lambda s: re.sub(r"\s+", " ", s).strip()


def close(t, i):  # index of the bracket closing the `{` or `(` at t[i]
    o, c, n = t[i], {"{": "}", "(": ")"}[t[i]], 0
    for j in range(i, len(t)):
        n += (t[j] == o) - (t[j] == c)
        if n == 0:
            return j


HEAD = re.compile(r"^[ \t]*fn (\w+)(<'a>)?\(([^)]*)\)\s*->\s*([^{]+?)\s*\{", re.M)
INST = r'\[\{ "pick": "instance", "instanceId": (.+?) \}\]'
HERO = r'\[\{ "pick": "hero", "player": (.+?) \}\]'
FORMS = [(r'\{ "targets": ' + INST + r" \}", "i", 1), (r'\{ "targets": ' + HERO + r" \}", "h", 1),
         (r'\{ "targets": pick::instance\((\w+)\) \}', "c", 1), (HERO, "h", 0)]


def pick_call(params, body):  # (prefix, suffix, zero-arg text) of the shared call
    m = re.fullmatch(r"json!\((.*)\)", squash(body))
    ps = [tuple(x.strip() for x in p.split(":", 1)) for p in params.split(",") if p.strip()]
    for rx, kind, wrap in FORMS if m else ():
        f = re.fullmatch(rx, m.group(1))
        if not f:
            continue
        x, ty = f.group(1), [p[1] for p in ps]
        if kind == "c" and ty == ["&CardInstance"] and x == ps[0][0]:
            r = ("pick::instance(", ")", None)
        elif kind == "i" and ty == ["&str"] and x == ps[0][0]:
            r = ("pick::id(", ")", None)
        elif kind == "i" and ty == ["&CardInstance"] and x == ps[0][0] + ".id":
            r = ("pick::instance(", ")", None)
        elif kind == "i" and ty == ["&Scenario", "&str"] and x == f"{ps[0][0]}.card({ps[1][0]}).id":
            r = ("pick::card(", ")", None)
        elif kind == "h" and not ps and x in ('"p1"', '"p2"'):
            r = ("", "", f"pick::hero({x})")
        elif kind == "h" and ty in (["&str"], ["PlayerId"]) and x in (ps[0][0], ps[0][0] + ".as_str()"):
            r = ("pick::hero(", ")", None)
        else:
            return None
        return (f"pick::targets({r[0]}", f"{r[1]})", f"pick::targets({r[2]})" if r[2] else None) if wrap else r


def lookup_mut(head, t):  # a state lookup: its mutability, or None

def drop_def(t, start, end):  # drop t[start:end] (a fn) with its `///` lines
    while t[t.rfind("\n", 0, start - 1) + 1 : start].lstrip().startswith("///"):
