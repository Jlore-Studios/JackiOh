````markdown
               targets: "--workspace --exclude jackioh-tools --tests"
               skip: ""
             - leg: tools
               key: tools
               targets: "-p jackioh-tools --tests"
               skip: "--skip _passes_with_every_random_pool_rotated"

               key: pool-probe
               targets: "-p jackioh-tools --test pool_probe"
               skip: ""
       env:
         CARGO_PROFILE_TEST_DEBUG: "0"
       steps:
         - uses: actions/checkout@v4
         - uses: ./.github/actions/setup

             node: "false"
             cache-key: daily-tests-${{ matrix.key }}

         - name: this leg's tests a pull request leaves to the daily run

           run: |

             cargo test $TARGETS --features jackioh-engine/testkit,jackioh-engine/ts -- --ignored $SKIP
   ```

  - The first run of each key compiles from scratch, the probe's nested `target/pool-probe` included (about 10 minutes here).
````
