   /// the newest major's line, from the last `PatchTier::Major` step of `history` through its newest. A closed
   /// play version has no row; a cut one keeps its id, card, name and `live`. With no major in `history` the
   /// line is all of it.
   pub fn major_line_versions(versions: &[PlayVersion], history: &[PatchStep]) -> Vec<PlayVersion> {
       let from = history.iter().rposition(|step| step.tier == PatchTier::Major).unwrap_or(0);
       let line: IndexSet<&str> = history[from..].iter().map(|step| step.version.as_str()).collect();
       versions
           .iter()
           .filter(|version| version.live)
           .filter_map(|version| {
               let cut: Vec<String> =
                   version.versions.iter().filter(|shipped| line.contains(shipped.as_str())).cloned().collect();
               (!cut.is_empty()).then(|| PlayVersion { versions: cut, ..version.clone() })
           })
           .collect()
   }
