The route `GET /api/stats/versions?scope=major` sends each live play version cut to the newest major's line.    pub fn major_line_versions(versions: &[PlayVersion], history: &[PatchStep]) -> Vec<PlayVersion> {
       let from = history.iter().rposition(|step| step.tier == PatchTier::Major).unwrap_or(0);
