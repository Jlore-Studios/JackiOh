````markdown
       pub set: &'static str,
       /// B2.2: its id segment (`classicplus` holds no hyphen).
       pub segment: &'static str,

       pub cards: usize,
       /// Non-token cards outside the set's numbers, indexed T-name (R1857001).
       pub shared_cards: &'static [&'static str],
       /// Tokens a card defines, indexed N.k after card N.
       pub card_defined_tokens: &'static [&'static str],
       /// Tokens no one card defines, indexed T-name.
       pub shared_tokens: &'static [&'static str],
       /// The rarity distribution over its non-token cards.
       pub rarities: &'static [(&'static str, usize)],
   }

   impl SetCensus {
       /// Its cards, numbered and not.
       pub const fn card_count(&self) -> usize { self.cards + self.shared_cards.len() }
       /// Its tokens.
       pub const fn token_count(&self) -> usize { self.card_defined_tokens.len() + self.shared_tokens.len() }
       /// R1420: whether the set ships; one that does not counts toward no total.
       pub fn ships(&self) -> bool {
           SetName::ALL.iter().find(|set| set.as_str() == self.set).is_some_and(|set| set_ships(*set))
       }
   }
   ```
````
