````markdown
"""#1125's cull. python3 -I cull.py delete|tidy <list>. Each list line is `<file>: <name> <name> ...`, the
file a unique prefix of a path under crates/cards/src/scripts/, each name an fn (a unique prefix of its
name), a CONST or `mod:<name>`; delete removes each with the attributes and comments right above it."""
import glob, re, sys

mode, listing = sys.argv[1], sys.argv[2]
ROOT = "crates/cards/src/scripts/"
LEX = re.compile(r'//[^\n]*|b?r(#*)"(?:.|\n)*?"\1|b?"(?:\\.|[^"\\])*"|\'(?:\\.|[^\\\'\n])\'|[{};]', re.S)

def end_of(src, at, const):
    depth = 0
    for m in LEX.finditer(src, at):
        t = m.group(0)
        if t == ";" and const and depth == 0:
            return m.end()
        depth += (t == "{") - (t == "}")
        if t == "}" and depth == 0 and not const:
            return m.end()
    sys.exit("unbalanced braces")

for line in open(listing, encoding="utf-8"):
    if not line.strip():
        continue
    ref, names = line.split(":", 1)
    paths = glob.glob(ROOT + ref.strip() + "*.rs")
    if len(paths) != 1:
        sys.exit(f"{ref}: {len(paths)} files")
    path = paths[0]
    src = open(path, encoding="utf-8").read()
    if mode == "tidy":
        src = re.sub(r"^[ \t]+\n", "", src, flags=re.M)
        src = re.sub(r"\n\n+", "\n\n", src)
        src = re.sub(r"\{\n\n", "{\n", src)
        src = re.sub(r"\n\n([ \t]*\})", r"\n\1", src)
    for name in names.split() if mode == "delete" else []:
        if name.startswith("mod:"):
            pattern = rf"^[ \t]*mod {name[4:]} \{{"
        elif name.isupper():
            pattern = rf"^[ \t]*const {name}\b"
        else:
            pattern = rf"^[ \t]*fn {name}[(<]"
        hits = list(re.finditer(pattern, src, re.M))
        if not hits and not name.startswith("mod:") and not name.isupper():
            hits = list(re.finditer(rf"^[ \t]*fn {name}\w*[(<]", src, re.M))
        if len(hits) != 1:
            sys.exit(f"{path}: {len(hits)} items match {name}")
        start = hits[0].start()
        above = src[:start].split("\n")[:-1]
        while above and above[-1].strip().startswith(("#[", "//")):
            start -= len(above.pop()) + 1
        end = end_of(src, hits[0].start(), name.isupper())
        end += src[end:].startswith("\n")
        if src[:start].endswith("\n\n") and src[end:].startswith("\n"):
            end += 1
        src = src[:start] + src[end:]
    open(path, "w", encoding="utf-8").write(src)
```
````
